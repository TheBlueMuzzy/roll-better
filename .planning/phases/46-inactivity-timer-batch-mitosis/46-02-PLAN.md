**Current flow (to be replaced):**
1. Player selects dice (tap) → presses UNLOCK button
2. handleConfirmUnlock() in App.tsx builds UnlockAnimation[] with findClearSpot()
3. setUnlockAnimations() triggers MitosisDie rendering
4. After animation duration: confirmUnlock() updates poolSize + pendingNewDice
5. clearUnlockAnimations() → startAIUnlockAnimations() → setPhase('idle')

**New flow:**
1. Player drags dice (committed on drop) — accumulated in committedUnlocks
2. Inactivity timer expires → onTimeout fires
3. Build UnlockAnimation[] from committedUnlocks (each committed die's position as fromPos)
4. Run mitosis animations
5. After duration: update poolSize + pendingNewDice from committedUnlocks
6. Clear committedUnlocks → AI unlocks → idle

**Key difference:** MitosisDie's `fromPos` was the lock slot position. Now it should be the committed die's position in the rolling zone (where the player dropped it). The die "splits" from where it sits, not from the lock row.

```typescript
const handleUnlockTimerExpire = useCallback(() => {
  const state = useGameStore.getState();
  const committed = state.committedUnlocks;

  // Must-unlock check: if poolSize=0 and lockedDice exist and 0 committed,
  // auto-commit one die (AFK-style)
  if (committed.length === 0) {
    const player = state.players[0];
    if (player.poolSize === 0 && player.lockedDice.length > 0 && player.lockedDice.length < 8) {
      // Must unlock at least 1 — auto-commit first locked die
      // Use a position in the rolling zone center
      const autoSlot = player.lockedDice[0];
      const autoPos: [number, number, number] = [ROLLING_X_OFFSET, DIE_SIZE / 2, 0];
      // Directly add to committedUnlocks and remove from lockedDice
      useGameStore.getState().forceCommitUnlock(autoSlot.goalSlotIndex, autoSlot.value, autoPos);
      // Re-read committed after force commit
      const updatedCommitted = useGameStore.getState().committedUnlocks;
      buildAndRunMitosis(updatedCommitted);
    } else {
      // Skip — 0 unlocks, go straight to AI unlocks → idle
      startAIUnlockAnimations();
    }
    return;
  }

  buildAndRunMitosis(committed);
}, [startAIUnlockAnimations]);
```

Create helper `buildAndRunMitosis(committed: CommittedUnlock[])`:

```typescript
const buildAndRunMitosis = useCallback((committed: CommittedUnlock[]) => {
  const state = useGameStore.getState();
  const existingPoolPositions = [...state.roundState.remainingDicePositions];
  const occupied: [number, number, number][] = [...existingPoolPositions];

  const allAnimations: UnlockAnimation[] = [];

  for (let i = 0; i < committed.length; i++) {
    const cu = committed[i];
    // fromPos = where the committed die sits in rolling zone
    const fromPos: [number, number, number] = [cu.position[0], DIE_SIZE / 2, cu.position[2]];

    // Find clear spot for split targets (avoids existing pool + previous splits)
    const { targetPos, splitTargets } = findClearSpot(occupied, DIE_SIZE);
    occupied.push(splitTargets[0], splitTargets[1]);

    const DEG30 = (30 * Math.PI) / 180;
    const prevDelay = allAnimations.length > 0 ? allAnimations[allAnimations.length - 1].delay : 0;
    const delay = allAnimations.length === 0 ? 0 : prevDelay + (0.25 + Math.random() * 0.25);

    allAnimations.push({
      slotIndex: cu.slotIndex,
      value: cu.value,
      fromPos,
      targetPos,
      splitTargets,
      splitYRotations: [
        (Math.random() * 2 - 1) * DEG30,
        (Math.random() * 2 - 1) * DEG30,
      ],
      delay,
    });
  }

  // Clear committed dice visuals BEFORE starting animations
  // (MitosisDie will render at fromPos, replacing the CommittedDie visuals)
  useGameStore.getState().clearCommittedUnlocks();

  if (allAnimations.length > 0) {
    useGameStore.getState().setUnlockAnimations(allAnimations);

    const lastDelay = allAnimations[allAnimations.length - 1].delay;
    const totalWait = (lastDelay * 1000) + 1800;

    setTimeout(() => {
      // Build pendingNewDice from committed unlocks (each → 2 dice)
      finalizeBatchUnlock(allAnimations);
      useGameStore.getState().clearUnlockAnimations();
      if (!isOnlineGame) {
        startAIUnlockAnimations();
      }
    }, totalWait);
  } else {
    startAIUnlockAnimations();
  }
}, [isOnlineGame, startAIUnlockAnimations]);
```

```typescript
function finalizeBatchUnlock(animations: UnlockAnimation[]) {
  const state = useGameStore.getState();
  const players = [...state.players];
  const player = { ...players[0] };

  const pendingNewDice: number[] = [];
  const pendingNewDicePositions: [number, number, number][] = [];
  const pendingNewDiceRotations: [number, number, number][] = [];

  for (const anim of animations) {
    pendingNewDice.push(anim.value, anim.value); // Each unlock → 2 dice
    pendingNewDicePositions.push(anim.splitTargets[0], anim.splitTargets[1]);

    const faceRot = getFaceUpRotation(anim.value);
    for (const yRot of anim.splitYRotations) {
      const faceQ = new Quaternion().setFromEuler(new Euler(faceRot[0], faceRot[1], faceRot[2]));
      const yQ = new Quaternion().setFromEuler(new Euler(0, yRot, 0));
      const combined = yQ.multiply(faceQ);
      const result = new Euler().setFromQuaternion(combined);
      pendingNewDiceRotations.push([result.x, result.y, result.z]);
    }
  }

  // lockedDice already reduced by completeDragUnlock — just update poolSize
  player.poolSize = player.poolSize + animations.length * 2;
  player.selectedForUnlock = [];
  players[0] = player;

  useGameStore.getState().set({
    players,
    roundState: {
      ...useGameStore.getState().roundState,
      pendingNewDice,
      pendingNewDicePositions,
      pendingNewDiceRotations,
    },
  });
}
```
